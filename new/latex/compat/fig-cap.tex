%% \myboxed{size}{stuff}
%% size - percent of textwidth: full, twothirds, half
%% stuff - what is inside
%%
\newcommand{\myboxed}[2]{%
  \vspace{-\bigskipamount}%
  \begin{center}%
    \fbox{%
      \setlength{\mylengtha}{\textwidth}%
      \addtolength{\mylengtha}{-2\fboxsep}%
      \addtolength{\mylengtha}{-2\fboxrule}%
      \ifthenelse{\equal{#1}{twothirds}}{\setlength{\mylengtha}{.75\mylengtha}}{}%
      \ifthenelse{\equal{#1}{half}}{\setlength{\mylengtha}{.50\mylengtha}}{}%
      \begin{minipage}{\mylengtha}%
        \setlength{\myheigtha}{.05\mylengtha}%
        \vspace{\myheigtha}%
        \begin{center}%
          #2%
        \end{center}%
        %% \vspace{\myheigtha}%
      \end{minipage}%
      }%
  \end{center}%
  \vspace{-\bigskipamount}%
  \vspace{-\smallskipamount}%
  }%

%% \mySboxed{size}{stuff}

  \vspace{-\bigskipamount}%
  \begin{center}%
    \fbox{%

        \end{center}%
        %% \vspace{\myheigtha}%

      }%
  \end{center}%
  \vspace{-\bigskipamount}%
  \vspace{-\smallskipamount}%
  }%

%% \mycapted{label}{stuff}{list-title}{caption-title}{text}
%%
%% label - the name for reference
%% stuff - what is capted
%% list-title - title that go in list of caption (same as
%%              caption-title if empty)
%% caption-title - title that go in the caption
%% text - commenting text
%%
\newcommand{\mycapted}[5]{%
  \settowidth{\mylengthb}{\emph{#4}}%
  \ifthenelse{\lengthtest{\mylengthb > .8\mylengtha}}%
  {\setlength{\mylengthb}{.8\mylengtha}}{}%
  \sbox{\myboxa}{\centering\parbox[t]{\mylengthb}{\protect\emph{#4}}}
  %% put the caption guts
  #2%
  \begin{center}%
    %% put the caption text
    \begin{minipage}{.9\mylengtha}%
      \addtolength{\baselineskip}{-0.11\baselineskip}%
      \begin{small}%
        #5%
      \end{small}%
    \end{minipage}%
    %% and the caption title
    \ifthenelse{\equal{#3}{}\and\equal{#1}{}}{\caption[#4]{\usebox{\myboxa}}}{}%
    \ifthenelse{\equal{#3}{}\and\not\equal{#1}{}}{\caption[#4]{\label{#1}\usebox{\myboxa}}}{}%
    \ifthenelse{\not\equal{#3}{}\and\equal{#1}{}}{\caption[#3]{\usebox{\myboxa}}}{}%
    \ifthenelse{\not\equal{#3}{}\and\not\equal{#1}{}}%
    {\caption[#3]{\label{#1}\usebox{\myboxa}}}{}%
    \smallskip%
  \end{center}%
  }%

%% \myScapted{stuff}{text}
%%
%% stuff - what is capted
%% text - commenting text
%%
\newcommand{\myScapted}[2]{%
  %% put the caption guts
  #1%
  %% \vspace{\myheigtha}%
  \smallskip
  \begin{center}%
    %% put the caption text
    \begin{minipage}{.95\mylengtha}%
      \addtolength{\baselineskip}{-0.22\baselineskip}%
      \begin{center}
        \begin{small}
          #2%
        \end{small}
      \end{center}
    \end{minipage}%
  \end{center}%
  %% \vspace{\myheigtha}%
  \smallskip
  }%

%% \myfigure{placement}{size}{type}{stuff}{list-title}{caption-title}{text}
%%
%% placement - first arg of \figure
%% size - \myboxed arg
%% type - directory prefix of the figure: dump, plot, fig
%% stuff - name of the figure file
%% list-title: title that go in list of caption
%% caption-title: title that go in the caption
%% text: commenting text
\newcommand{\myfigure}[7]{%
  \label{#3-#4-section}%
  \begin{figure}[#1]%
    \myboxed{#2}{%
      \mycapted{#3-#4}%
      {\historicalimage{file=../#3/#4,width=.9\mylengtha}}%
      {#5}{#6}{#7}%
      }%
  \end{figure}%
  }%

%% \mySfigure{size}{type}{stuff}{text}
%%
%% size - \mySboxed arg
%% type - directory prefix of the figure: dump, plot, fig
%% stuff - name of the figure file
%% text: commenting text
\newcommand{\mySfigure}[4]{%
    \mySboxed{#1}{\myScapted{\historicalimage{file=../#2/#3,width=.95\mylengtha}}{#4}}}

%% \mytwofigure{placement}{size}{type}{HV}{stuff1}{stuff2}{list-title}{caption-title}{text}
%%              #1         #2    #3    #4  #5      #6      #7          #8             #9  
%%
%% placement - first arg of \figure
%% size - \myboxed arg
%% type - directory prefix of the figure: dump, plot, fig
%% stuff1 - name of the figure file 1
%% stuff2 - name of the figure file 2
%% HV - hor-izontal or vert-ical
%% list-title: title that go in list of caption
%% caption-title: title that go in the caption
%% text: commenting text
\newcommand{\mytwofigure}[9]{%
  \label{#3-#5+#6-section}%
  \begin{figure}[#1]%
    \myboxed{#2}{%
      \mycapted{#3-#5+#6}%
      {\ifthenelse{\equal{#4}{hor}}%
      {\historicalimage{file=../#3/#5,width=.45\mylengtha}%
        \historicalimage{file=../#3/#6,width=.45\mylengtha}}%
      {\historicalimage{file=../#3/#5,width=.9\mylengtha} \par%
        \historicalimage{file=../#3/#6,width=.9\mylengtha}}}%
      {#7}{#8}{#9}%
      }%
  \end{figure}%
  }%
