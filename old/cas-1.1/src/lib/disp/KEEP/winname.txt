{
	char* window_name = "foo";
	XSizeHints size_hints;
	XWMHints wm_hints;
	XClassHint class_hints;
	XTextProperty windowName;

	size_hints.flags = /* PPosition | PSize | */ PMinSize | PResizeInc;
	size_hints.min_width = 256;
	size_hints.min_height = 256;
	size_hints.width_inc = width;
	size_hints.height_inc = height;

	if (XStringListToTextProperty(&window_name, 1, &windowName) == 0) {
		fprintf(stderr, "%s : structure allocation for windowName failed\n",
			cmd_name);
		exit(1);
	}

	wm_hints.initial_state = NormalState;
	wm_hints.input = True;
	wm_hints.flags = StateHint | InputHint;
	class_hints.res_name = cmd_name;
	class_hints.res_class = "Foo";
	XSetWMProperties(display, window, &windowName, NULL, argv, argc,
			 &size_hints, &wm_hints, &class_hints);
}
